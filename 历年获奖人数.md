<!-- ===== ① 动态加载 ECharts：本地优先 + 多 CDN 回退 ===== -->
<script>
(function () {
  var sources = [
    '/assets/js/echarts.min.js',
    'https://cdn.bootcdn.net/ajax/libs/echarts/5.5.1/echarts.min.js',
    'https://lib.baomitu.com/echarts/5.5.1/echarts.min.js',
    'https://cdn.jsdelivr.net/npm/echarts@5/dist/echarts.min.js',
    'https://unpkg.com/echarts@5/dist/echarts.min.js'
  ];
  window.__echartsReady = new Promise(function (resolve, reject) {
    if (window.echarts) { resolve(); return; }
    var i = 0;
    function next() {
      if (i >= sources.length) {
        reject(new Error('ECharts 所有源均加载失败，请检查网络/代理，或下载 echarts.min.js 到 assets/js/ 目录'));
        return;
      }
      var s = document.createElement('script');
      s.src = sources[i++];
      s.onload = function () { window.echarts ? resolve() : next(); };
      s.onerror = next;
      document.head.appendChild(s);
    }
    next();
  });
})();
</script>
